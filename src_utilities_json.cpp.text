namespace rms::json {

Value Value::makeBool(bool v) { Value x; x.type = Type::Bool; x.boolean = v; return x; }
Value Value::makeNumber(double v) { Value x; x.type = Type::Number; x.number = v; return x; }
Value Value::makeString(std::string v) { Value x; x.type = Type::String; x.string = std::move(v); return x; }
Value Value::makeArray() { Value x; x.type = Type::Array; return x; }
Value Value::makeObject() { Value x; x.type = Type::Object; return x; }

const Value* Value::find(const std::string& key) const {
    if (type != Type::Object) return nullptr;
    for (const auto& kv : object)
        if (kv.first == key) return &kv.second;
    return nullptr;
}

Value* Value::find(const std::string& key) {
    if (type != Type::Object) return nullptr;
    for (auto& kv : object)
        if (kv.first == key) return &kv.second;
    return nullptr;
}

void Value::set(const std::string& key, Value v) {
    if (type != Type::Object) { *this = makeObject(); }
    if (Value* existing = find(key)) { *existing = std::move(v); return; }
    object.emplace_back(key, std::move(v));
}

void Value::push(Value v) {
    if (type != Type::Array) { *this = makeArray(); }
    array.push_back(std::move(v));
}

namespace {

constexpr int kMaxDepth = 64;

class Parser {
public:
    explicit Parser(const std::string& text) : s_(text) {}

    bool run(Value& out, std::string& error) {
        skipWs();
        if (!parseValue(out, 0)) {
            error = error_ + " (posición " + std::to_string(pos_) + ")";
            return false;
        }
        skipWs();
        if (pos_ != s_.size()) {
            error = "Contenido inesperado tras el valor JSON (posición " + std::to_string(pos_) + ")";
            return false;
        }
        return true;
    }

private:
    const std::string& s_;
    size_t pos_ = 0;
    std::string error_;

    bool fail(const char* msg) { error_ = msg; return false; }

    static bool isDigit(char c) { return c >= '0' && c <= '9'; }

    void skipWs() {
        while (pos_ < s_.size()) {
            char c = s_[pos_];
            if (c == ' ' || c == '\t' || c == '\n' || c == '\r') ++pos_;
            else break;
        }
    }

    bool literal(const char* lit) {
        size_t n = std::strlen(lit);
        if (s_.compare(pos_, n, lit) != 0) return fail("Literal no válido");
        pos_ += n;
        return true;
    }

    bool parseValue(Value& v, int depth) {
        if (depth > kMaxDepth) return fail("Anidamiento JSON excesivo");
        skipWs();
        if (pos_ >= s_.size()) return fail("Fin de archivo inesperado");
        char c = s_[pos_];
        switch (c) {
        case '{': return parseObject(v, depth);
        case '[': return parseArray(v, depth);
        case '"': v = Value::makeString({}); return parseString(v.string);
        case 't': v = Value::makeBool(true); return literal("true");
        case 'f': v = Value::makeBool(false); return literal("false");
        case 'n': v = Value(); return literal("null");
        default:
            if (c == '-' || isDigit(c)) return parseNumber(v);
            return fail("Carácter inesperado");
        }
    }

    bool parseObject(Value& v, int depth) {
        v = Value::makeObject();
        ++pos_;
        skipWs();
        if (pos_ < s_.size() && s_[pos_] == '}') { ++pos_; return true; }
        for (;;) {
            skipWs();
            if (pos_ >= s_.size() || s_[pos_] != '"') return fail("Se esperaba una clave entre comillas");
            std::string key;
            if (!parseString(key)) return false;
            skipWs();
            if (pos_ >= s_.size() || s_[pos_] != ':') return fail("Se esperaba ':'");
            ++pos_;
            Value child;
            if (!parseValue(child, depth + 1)) return false;
            if (v.find(key) != nullptr) return fail("Clave duplicada en objeto");
            v.object.emplace_back(std::move(key), std::move(child));
            skipWs();
            if (pos_ >= s_.size()) return fail("Objeto sin cerrar");
            if (s_[pos_] == ',') { ++pos_; continue; }
            if (s_[pos_] == '}') { ++pos_; return true; }
            return fail("Se esperaba ',' o '}'");
        }
    }

    bool parseArray(Value& v, int depth) {
        v = Value::makeArray();
        ++pos_;
        skipWs();
        if (pos_ < s_.size() && s_[pos_] == ']') { ++pos_; return true; }
        for (;;) {
            Value child;
            if (!parseValue(child, depth + 1)) return false;
            v.array.push_back(std::move(child));
            skipWs();
            if (pos_ >= s_.size()) return fail("Array sin cerrar");
            if (s_[pos_] == ',') { ++pos_; continue; }
            if (s_[pos_] == ']') { ++pos_; return true; }
            return fail("Se esperaba ',' o ']'");
        }
    }

    bool parseHex4(uint32_t& out) {
        if (pos_ + 4 > s_.size()) return fail("Escape \\u incompleto");
        out = 0;
        for (int i = 0; i < 4; ++i) {
            char c = s_[pos_++];
            out <<= 4;
            if (c >= '0' && c <= '9') out |= static_cast<uint32_t>(c - '0');
            else if (c >= 'a' && c <= 'f') out |= static_cast<uint32_t>(c - 'a' + 10);
            else if (c >= 'A' && c <= 'F') out |= static_cast<uint32_t>(c - 'A' + 10);
            else return fail("Dígito hexadecimal no válido");
        }
        return true;
    }

    static void appendUtf8(std::string& o, uint32_t cp) {
        if (cp < 0x80) {
            o.push_back(static_cast<char>(cp));
        } else if (cp < 0x800) {
            o.push_back(static_cast<char>(0xC0 | (cp >> 6)));
            o.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
        } else if (cp < 0x10000) {
            o.push_back(static_cast<char>(0xE0 | (cp >> 12)));
            o.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
            o.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
        } else {
            o.push_back(static_cast<char>(0xF0 | (cp >> 18)));
            o.push_back(static_cast<char>(0x80 | ((cp >> 12) & 0x3F)));
            o.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
            o.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
        }
    }

    bool parseString(std::string& out) {
        ++pos_; // opening quote
        out.clear();
        for (;;) {
            if (pos_ >= s_.size()) return fail("Cadena sin cerrar");
            unsigned char c = static_cast<unsigned char>(s_[pos_++]);
            if (c == '"') return true;
            if (c < 0x20) return fail("Carácter de control dentro de una cadena");
            if (c != '\\') { out.push_back(static_cast<char>(c)); continue; }
            if (pos_ >= s_.size()) return fail("Escape incompleto");
            char e = s_[pos_++];
            switch (e) {
            case '"': out.push_back('"'); break;
            case '\\': out.push_back('\\'); break;
            case '/': out.push_back('/'); break;
            case 'b': out.push_back('\b'); break;
            case 'f': out.push_back('\f'); break;
            case 'n': out.push_back('\n'); break;
            case 'r': out.push_back('\r'); break;
            case 't': out.push_back('\t'); break;
            case 'u': {
                uint32_t cp = 0;
                if (!parseHex4(cp)) return false;
                if (cp >= 0xD800 && cp <= 0xDBFF) {
                    if (pos_ + 2 > s_.size() || s_[pos_] != '\\' || s_[pos_ + 1] != 'u')
                        return fail("Par sustituto UTF-16 incompleto");
                    pos_ += 2;
                    uint32_t lo = 0;
                    if (!parseHex4(lo)) return false;
                    if (lo < 0xDC00 || lo > 0xDFFF) return fail("Par sustituto UTF-16 no válido");
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
                } else if (cp >= 0xDC00 && cp <= 0xDFFF) {
                    return fail("Sustituto UTF-16 aislado");
                }
                appendUtf8(out, cp);
                break;
            }
            default:
                return fail("Secuencia de escape no válida");
            }
        }
    }

    bool parseNumber(Value& v) {
        const size_t start = pos_;
        if (s_[pos_] == '-') ++pos_;
        if (pos_ >= s_.size()) return fail("Número incompleto");
        if (s_[pos_] == '0') {
            ++pos_;
        } else if (isDigit(s_[pos_])) {
            while (pos_ < s_.size() && isDigit(s_[pos_])) ++pos_;
        } else {
            return fail("Número no válido");
        }
        if (pos_ < s_.size() && s_[pos_] == '.') {
            ++pos_;
            if (pos_ >= s_.size() || !isDigit(s_[pos_])) return fail("Decimal incompleto");
            while (pos_ < s_.size() && isDigit(s_[pos_])) ++pos_;
        }
        if (pos_ < s_.size() && (s_[pos_] == 'e' || s_[pos_] == 'E')) {
            ++pos_;
            if (pos_ < s_.size() && (s_[pos_] == '+' || s_[pos_] == '-')) ++pos_;
            if (pos_ >= s_.size() || !isDigit(s_[pos_])) return fail("Exponente incompleto");
            while (pos_ < s_.size() && isDigit(s_[pos_])) ++pos_;
        }
        double d = 0.0;
        auto res = std::from_chars(s_.data() + start, s_.data() + pos_, d);
        if (res.ec != std::errc() || !std::isfinite(d)) return fail("Número fuera de rango");
        v = Value::makeNumber(d);
        return true;
    }
};

void writeEscaped(std::string& out, const std::string& s) {
    out.push_back('"');
    for (unsigned char c : s) {
        switch (c) {
        case '"': out += "\\\""; break;
        case '\\': out += "\\\\"; break;
        case '\b': out += "\\b"; break;
        case '\f': out += "\\f"; break;
        case '\n': out += "\\n"; break;
        case '\r': out += "\\r"; break;
        case '\t': out += "\\t"; break;
        default:
            if (c < 0x20) {
                static const char* hex = "0123456789abcdef";
                out += "\\u00";
                out.push_back(hex[c >> 4]);
                out.push_back(hex[c & 0xF]);
            } else {
                out.push_back(static_cast<char>(c));
            }
        }
    }
    out.push_back('"');
}

void writeNumber(std::string& out, double d) {
    if (!std::isfinite(d)) { out += "0"; return; }
    if (std::floor(d) == d && std::fabs(d) < 9.0e15) {
        out += std::to_string(static_cast<long long>(d));
        return;
    }
    char buf[64];
    auto res = std::to_chars(buf, buf + sizeof(buf), d);
    if (res.ec == std::errc()) out.append(buf, res.ptr);
    else out += "0";
}

void newline(std::string& out, int indent, int level) {
    if (indent <= 0) return;
    out.push_back('\n');
    out.append(static_cast<size_t>(indent * level), ' ');
}

void writeValue(std::string& out, const Value& v, int indent, int level) {
    switch (v.type) {
    case Value::Type::Null: out += "null"; break;
    case Value::Type::Bool: out += v.boolean ? "true" : "false"; break;
    case Value::Type::Number: writeNumber(out, v.number); break;
    case Value::Type::String: writeEscaped(out, v.string); break;
    case Value::Type::Array:
        if (v.array.empty()) { out += "[]"; break; }
        out.push_back('[');
        for (size_t i = 0; i < v.array.size(); ++i) {
            if (i) out.push_back(',');
            newline(out, indent, level + 1);
            writeValue(out, v.array[i], indent, level + 1);
        }
        newline(out, indent, level);
        out.push_back(']');
        break;
    case Value::Type::Object:
        if (v.object.empty()) { out += "{}"; break; }
        out.push_back('{');
        for (size_t i = 0; i < v.object.size(); ++i) {
            if (i) out.push_back(',');
            newline(out, indent, level + 1);
            writeEscaped(out, v.object[i].first);
            out += indent > 0 ? ": " : ":";
            writeValue(out, v.object[i].second, indent, level + 1);
        }
        newline(out, indent, level);
        out.push_back('}');
        break;
    }
}

} // namespace

bool parse(const std::string& text, Value& out, std::string& error) {
    Parser p(text);
    Value tmp;
    if (!p.run(tmp, error)) return false;
    out = std::move(tmp);
    return true;
}

std::string serialize(const Value& v, int indent) {
    std::string out;
    writeValue(out, v, indent, 0);
    if (indent > 0) out.push_back('\n');
    return out;
}

} // namespace rms::json
