#pragma once
// Common Windows include + UTF-8/UTF-16 conversion + stable application paths.

#ifndef WIN32_LEAN_AND_MEAN
#define WIN32_LEAN_AND_MEAN
#endif
#ifndef NOMINMAX
#define NOMINMAX
#endif
#include <windows.h>

#include <filesystem>
#include <string>

namespace rms::platform {

inline std::wstring utf8ToWide(const std::string& s) {
    if (s.empty()) return {};
    int n = MultiByteToWideChar(CP_UTF8, 0, s.data(), static_cast<int>(s.size()), nullptr, 0);
    if (n <= 0) return {};
    std::wstring out(static_cast<size_t>(n), L'\0');
    MultiByteToWideChar(CP_UTF8, 0, s.data(), static_cast<int>(s.size()), out.data(), n);
    return out;
}

inline std::string wideToUtf8(const std::wstring& w) {
    if (w.empty()) return {};
    int n = WideCharToMultiByte(CP_UTF8, 0, w.data(), static_cast<int>(w.size()), nullptr, 0, nullptr, nullptr);
    if (n <= 0) return {};
    std::string out(static_cast<size_t>(n), '\0');
    WideCharToMultiByte(CP_UTF8, 0, w.data(), static_cast<int>(w.size()), out.data(), n, nullptr, nullptr);
    return out;
}

// Directory containing the running executable (never the working directory).
inline std::filesystem::path appDirectory() {
    std::wstring buf(MAX_PATH, L'\0');
    for (;;) {
        DWORD n = GetModuleFileNameW(nullptr, buf.data(), static_cast<DWORD>(buf.size()));
        if (n == 0) {
            std::error_code ec;
            return std::filesystem::current_path(ec);
        }
        if (n < buf.size()) {
            buf.resize(n);
            break;
        }
        if (buf.size() >= 32768) {
            std::error_code ec;
            return std::filesystem::current_path(ec);
        }
        buf.resize(buf.size() * 2);
    }
    return std::filesystem::path(buf).parent_path();
}

inline std::filesystem::path macrosDirectory() { return appDirectory() / L"macros"; }
inline std::filesystem::path configDirectory() { return appDirectory() / L"config"; }
inline std::filesystem::path settingsFilePath() { return configDirectory() / L"settings.json"; }

} // namespace rms::platform
