#pragma once
// Movement calibration.
//   real = round((base / sensitivity) * (monitorWidth / 1280))
// Width only (never height or pixel count). Rounding: half away from zero (std::round).
// Result is clamped to +/- kMaxScaledMove to prevent absurd jumps.
// This does NOT guarantee identical in-game rotation across all Roblox setups.

#include <algorithm>
#include <cmath>

namespace rms::calibration {

constexpr double kDefaultSensitivity = 0.74;
constexpr double kMinSensitivity = 0.01;
constexpr double kMaxSensitivity = 20.0;
constexpr int kReferenceWidth = 1280;
constexpr int kReferenceHeight = 960;
constexpr int kMaxScaledMove = 10000;

inline bool isValidSensitivity(double s) {
    return std::isfinite(s) && s >= kMinSensitivity && s <= kMaxSensitivity;
}

inline int scaleMovement(int base, double sensitivity, int monitorWidth) {
    if (!isValidSensitivity(sensitivity)) sensitivity = kDefaultSensitivity;
    if (monitorWidth <= 0) monitorWidth = kReferenceWidth;
    double v = (static_cast<double>(base) / sensitivity) *
               (static_cast<double>(monitorWidth) / static_cast<double>(kReferenceWidth));
    v = std::round(v);
    v = std::clamp(v, static_cast<double>(-kMaxScaledMove), static_cast<double>(kMaxScaledMove));
    return static_cast<int>(v);
}

} // namespace rms::calibration
